Fix critical lock conflict bug in src/lib/supabase.ts on SkillSnap (skillsnap.com.au):

━━━━━━━━━━━━━━━━━━━━━━━
PROBLEM
━━━━━━━━━━━━━━━━━━━━━━━
The app is throwing this error constantly across multiple components:
"Lock 'lock:sb-dnraeyxjzdmpdvrkzyfd-auth-token' was released because another request stole it"

This is caused by @supabase/ssr's createBrowserClient using the Web Lock API for cookie access. When multiple concurrent Supabase requests fire simultaneously (getSession, hydrateProfile, loadThreads, featured profiles query), they compete for the same lock and steal it from each other — causing auth failures, session drops, and components showing sign-in screens.

━━━━━━━━━━━━━━━━━━━━━━━
FIX — replace createBrowserClient with createClient
━━━━━━━━━━━━━━━━━━━━━━━
The server-side callback route (src/app/auth/callback/route.ts) already handles the PKCE exchange using @supabase/ssr on the server. The browser client does NOT need @supabase/ssr — it just needs a stable regular client that uses localStorage for session storage (no Web Lock API, no cookie conflicts).

Replace the entire contents of src/lib/supabase.ts with this:

import { createClient } from "@supabase/supabase-js";

type SupabaseClient = ReturnType;

const REAL_URL = process.env.NEXT_PUBLIC_SUPABASE_URL!;
const ANON_KEY = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!;

function isOrchidsSandbox(): boolean {
  return (
    typeof window !== "undefined" &&
    window.location.hostname.includes("orchids.cloud")
  );
}

// Single singleton client for the entire app lifetime.
// Uses standard createClient (localStorage-based) — no Web Lock API,
// no cookie conflicts. The server-side callback route handles PKCE exchange.
let _client: SupabaseClient | null = null;

export function getSupabase(): SupabaseClient {
  if (!_client) {
    const url =
      typeof window !== "undefined" && isOrchidsSandbox()
        ? `${window.location.origin}/api/proxy`
        : REAL_URL;
    _client = createClient(url, ANON_KEY, {
      auth: {
        persistSession: true,
        storageKey: "sb-skillsnap-auth-token",
        storage: typeof window !== "undefined" ? window.localStorage : undefined,
        autoRefreshToken: true,
        detectSessionInUrl: true,
        flowType: "pkce",
      },
    });
  }
  return _client;
}

export function getAuthSupabase(): SupabaseClient {
  return getSupabase();
}

export function getRealtimeSupabase(): SupabaseClient {
  return getSupabase();
}

export function resetSupabaseClient() {
  _client = null;
}
