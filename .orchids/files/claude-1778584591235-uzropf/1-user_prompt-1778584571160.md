In AppState.tsx, the profile fetch inside hydrateProfile hangs on second tab because the Supabase auth lock blocks it. Fix by using getSupabase() (not getAuthSupabase()) for the profile fetch, and use getUser() instead of relying on the session from onAuthStateChange. Replace the fetch inside hydrateProfile with:
typescriptasync function hydrateProfile(userId: string, authUser: import("@supabase/supabase-js").User) {
  const timeoutId = setTimeout(() => {
    console.error("[AppState] hydrateProfile timed out");
    dispatch({ type: "SET_AUTH_LOADING", loading: false });
  }, 5000);

  try {
    const sb = getAuthSupabase();
    const { data, error: fetchErr } = await sb
      .from("profiles")
      .select("*")
      .eq("id", userId)
      .single();

    clearTimeout(timeoutId);
    console.log("[AppState] hydrateProfile fetch complete", !!data, fetchErr?.message);

    if (data) {
      const mappedUser = mapProfile(data as Record<string, unknown>);
      dispatch({ type: "SET_AUTH", user: mappedUser });
      return;
    }

    if (fetchErr && fetchErr.code !== "PGRST116") {
      dispatch({ type: "SET_AUTH_LOADING", loading: false });
      return;
    }

    const user = await ensureProfile(authUser);
    if (user) {
      dispatch({ type: "SET_AUTH", user });
    } else {
      dispatch({ type: "SET_AUTH_LOADING", loading: false });
    }
  } catch (e) {
    clearTimeout(timeoutId);
    console.error("[AppState] hydrateProfile CRASHED:", e);
    dispatch({ type: "SET_AUTH_LOADING", loading: false });
  }
}
Also in supabase.ts, change flowType from "pkce" to "implicit" — PKCE flow requires a code exchange which causes lock contention on multiple tabs. Implicit flow uses the token directly from localStorage without needing a lock.
